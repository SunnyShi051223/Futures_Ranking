N:=4;
STEP1:=2/100;
MVALUE1:=20/100;

// ===============================
// SAR 计算
// ===============================
SARRAW:=SAR(N,STEP1,MVALUE1);
SARLINE:ABS(SARRAW),CIRCLEDOT;

// ===============================
// 方向显示
// 1=多头，-1=空头
// ===============================
DIR_SHOW:IF(SARRAW>0,1,-1);

// ===============================
// SAR 状态持仓型：纯做多
// ===============================

// SAR 多头状态：开多/持多
SARRAW>0,BK;

// SAR 空头状态：平多
SARRAW<0,SP;

AUTOFILTER;
