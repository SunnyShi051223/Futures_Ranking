N1:=12;
N2:=26;
M:=9;

// ===============================
// MACD 计算
// ===============================
DIFV:=EMA(CLOSE,N1)-EMA(CLOSE,N2);
DEAV:=EMA(DIFV,M);
MACDV:=2*(DIFV-DEAV);

// ===============================
// 指标线显示
// ===============================
DIFLINE:DIFV;
DEALINE:DEAV;
MACDLINE:MACDV;

// ===============================
// 纯做多逻辑
// ===============================
DIFV>DEAV,BK;
DIFV<DEAV,SP;

AUTOFILTER;
